cmake_minimum_required(VERSION 3.20)
project(wiz_protocol_checks C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
get_filename_component(REPO "${CMAKE_CURRENT_LIST_DIR}/../../../.." ABSOLUTE)
set(IDF_MBEDTLS "${REPO}/.tools/esp-idf/components/mbedtls")
set(ENABLE_PROGRAMS OFF CACHE BOOL "" FORCE)
set(ENABLE_TESTING OFF CACHE BOOL "" FORCE)
set(GEN_FILES OFF CACHE BOOL "" FORCE)
set(MBEDTLS_FATAL_WARNINGS OFF CACHE BOOL "" FORCE)
set(MBEDTLS_USER_CONFIG_FILE "${CMAKE_CURRENT_LIST_DIR}/mbedtls_user_config.h" CACHE FILEPATH "" FORCE)
include_directories(BEFORE "${REPO}/tests/native/tls_bundle/stubs" "${IDF_MBEDTLS}/port/include")
add_subdirectory("${IDF_MBEDTLS}/mbedtls" mbedtls)
add_library(wiz_host STATIC ../wiz_protocol.c ../wiz_capabilities.c ../wiz_crypto.c ../wiz_driver.c
    "${REPO}/managed_components/espressif__cjson/cJSON/cJSON.c")
if(APPLE)
    # The pinned upstream cJSON still uses sprintf in its unused print helpers.
    # Keep strict warnings for our code without patching the vendor checkout.
    set_source_files_properties("${REPO}/managed_components/espressif__cjson/cJSON/cJSON.c"
        PROPERTIES COMPILE_OPTIONS "-Wno-deprecated-declarations")
endif()
target_include_directories(wiz_host PUBLIC ../include "${REPO}/managed_components/espressif__cjson/cJSON")
target_compile_definitions(wiz_host PRIVATE CJSON_NESTING_LIMIT=6)
target_compile_options(wiz_host PRIVATE -Wall -Wextra -Werror -Wshadow)
target_link_libraries(wiz_host PUBLIC tfpsacrypto)
add_executable(wiz_checks test_wiz.c)
target_compile_definitions(wiz_checks PRIVATE WIZ_MODEL_FIXTURE="${CMAKE_CURRENT_LIST_DIR}/fixtures/esp25_shrgb_01_1.38.0_model.json")
target_compile_options(wiz_checks PRIVATE -Wall -Wextra -Werror -Wshadow -UNDEBUG)
target_link_libraries(wiz_checks PRIVATE wiz_host)
add_executable(wiz_fixture_checks test_fixtures.c)
target_compile_options(wiz_fixture_checks PRIVATE -Wall -Wextra -Werror -Wshadow)
target_link_libraries(wiz_fixture_checks PRIVATE wiz_host)
option(WIZ_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
if(WIZ_SANITIZERS)
    foreach(target wiz_host wiz_checks wiz_fixture_checks)
        target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(${target} PUBLIC -fsanitize=address,undefined)
    endforeach()
endif()
enable_testing()
add_test(NAME wiz_checks COMMAND wiz_checks)
