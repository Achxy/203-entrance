cmake_minimum_required(VERSION 3.16)
project(tls_bundle_regression C)
get_filename_component(PROJECT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(IDF_MBEDTLS "${PROJECT_ROOT}/.tools/esp-idf/components/mbedtls")
set(ENABLE_PROGRAMS OFF CACHE BOOL "" FORCE)
set(ENABLE_TESTING OFF CACHE BOOL "" FORCE)
set(GEN_FILES OFF CACHE BOOL "" FORCE)
set(MBEDTLS_FATAL_WARNINGS OFF CACHE BOOL "" FORCE)
set(MBEDTLS_USER_CONFIG_FILE "${CMAKE_CURRENT_SOURCE_DIR}/mbedtls_user_config.h" CACHE FILEPATH "" FORCE)
# The pinned ESP-IDF adds MPI/ECP wrappers; hardware acceleration remains off
# in this host build. No crypto, parsing or verification functions are mocked.
include_directories(BEFORE "${CMAKE_CURRENT_SOURCE_DIR}/stubs" "${IDF_MBEDTLS}/port/include")
add_subdirectory("${IDF_MBEDTLS}/mbedtls" mbedtls)
foreach(mode IN ITEMS legacy cross_signed)
 add_executable(${mode} check.c)
 target_include_directories(${mode} PRIVATE "${IDF_MBEDTLS}/esp_crt_bundle" "${IDF_MBEDTLS}/esp_crt_bundle/include")
 target_compile_options(${mode} PRIVATE -UNDEBUG)
 target_link_libraries(${mode} PRIVATE mbedtls mbedx509 tfpsacrypto)
endforeach()
target_compile_definitions(cross_signed PRIVATE CONFIG_MBEDTLS_CERTIFICATE_BUNDLE_CROSS_SIGNED_VERIFY=1)
